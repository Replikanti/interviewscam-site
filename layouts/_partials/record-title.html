{{- /* The fixed title of a record page or a section page (SPEC section 5, ADR-0004 part 4). Called with a dict and returns the title: page the page html true: HTML for the h1 (the persona label in its span); false: plain text, for and JSON-LD head optional, plain text only: true gives the <title> form. For a visible persona that is title.persona_head with the label first (SPEC section 5, ADR-0004 part 4); for every other page it is the title above The <title> block appends " · interviewscam.com". */ -}} {{- if and .head .html }}{{ errorf "record-title.html: head is plain text only, never html" }}{{ end -}} {{- $page := .page -}} {{- $out := "" -}} {{- if eq $page.Kind "section" -}} {{- $out = partial "label.html" (printf "title.%s" $page.Section) -}} {{- if .html }}{{ $out = htmlEscape $out }}{{ end -}} {{- else if and (eq $page.Section "reports") (eq $page.Params.name_type "persona") $page.Params.hide_name -}} {{- $out = partial "label.html" "title.hidden" -}} {{- if .html }}{{ $out = htmlEscape $out }}{{ end -}} {{- else if and .head (eq $page.Section "reports") (eq $page.Params.name_type "persona") -}} {{- $name := partial "record-name.html" (dict "page" $page "html" false "first" true) -}} {{- $out = printf (partial "label.html" "title.persona_head") $name -}} {{- else -}} {{- $format := partial "label.html" (cond (eq $page.Section "campaigns") "title.campaign" "title.report") -}} {{- $name := partial "record-name.html" (dict "page" $page "html" .html) -}} {{- if .html -}} {{- $out = printf (htmlEscape $format) $name -}} {{- else -}} {{- $out = printf $format $name -}} {{- end -}} {{- end -}} {{- if .html }}{{ $out = $out | safeHTML }}{{ end -}} {{- return $out -}}
